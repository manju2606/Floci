VPC

List VPCs

aws ec2 describe-vpcs \ --endpoint-url http://localhost:4566

Create VPC

aws ec2 create-vpc \ --cidr-block 10.0.0.0/16 \ --endpoint-url http://localhost:4566

Delete VPC

aws ec2 delete-vpc \ --vpc-id <vpc-id> \ --endpoint-url http://localhost:4566

Tag VPC

aws ec2 create-tags \ --resources <vpc-id> \ --tags Key=Name,Value=my-vpc \ --endpoint-url http://localhost:4566

Subnets

List subnets

aws ec2 describe-subnets \ --endpoint-url http://localhost:4566

Create subnet

aws ec2 create-subnet \ --vpc-id <vpc-id> \ --cidr-block 10.0.1.0/24 \ --availability-zone us-east-1a \ --endpoint-url http://localhost:4566

Enable public IP on subnet

aws ec2 modify-subnet-attribute \ --subnet-id <subnet-id> \ --map-public-ip-on-launch \ --endpoint-url http://localhost:4566

Delete subnet

aws ec2 delete-subnet \ --subnet-id <subnet-id> \ --endpoint-url http://localhost:4566

Internet Gateways

Create IGW

aws ec2 create-internet-gateway \ --endpoint-url http://localhost:4566

Attach IGW to VPC

aws ec2 attach-internet-gateway \ --internet-gateway-id <igw-id> \ --vpc-id <vpc-id> \ --endpoint-url http://localhost:4566

Detach IGW

aws ec2 detach-internet-gateway \ --internet-gateway-id <igw-id> \ --vpc-id <vpc-id> \ --endpoint-url http://localhost:4566

List IGWs

aws ec2 describe-internet-gateways \ --endpoint-url http://localhost:4566

Route Tables

List route tables

aws ec2 describe-route-tables \ --endpoint-url http://localhost:4566

Create route table

aws ec2 create-route-table \ --vpc-id <vpc-id> \ --endpoint-url http://localhost:4566

Add route to IGW

aws ec2 create-route \ --route-table-id <rtb-id> \ --destination-cidr-block 0.0.0.0/0 \ --gateway-id <igw-id> \ --endpoint-url http://localhost:4566

Associate subnet with RT

aws ec2 associate-route-table \ --route-table-id <rtb-id> \ --subnet-id <subnet-id> \ --endpoint-url http://localhost:4566
