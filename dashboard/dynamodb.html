Table Operations

Create table

aws dynamodb create-table \ --table-name MyTable \ --attribute-definitions \ AttributeName=Id,AttributeType=S \ --key-schema \ AttributeName=Id,KeyType=HASH \ --billing-mode PAY_PER_REQUEST \ --endpoint-url http://localhost:4566

List tables

aws dynamodb list-tables \ --endpoint-url http://localhost:4566

Describe table

aws dynamodb describe-table \ --table-name MyTable \ --endpoint-url http://localhost:4566

Delete table

aws dynamodb delete-table \ --table-name MyTable \ --endpoint-url http://localhost:4566

Item Operations

Put item

aws dynamodb put-item \ --table-name MyTable \ --item '{"Id":{"S":"1"},"Name":{"S":"Alice"}}' \ --endpoint-url http://localhost:4566

Get item

aws dynamodb get-item \ --table-name MyTable \ --key '{"Id":{"S":"1"}}' \ --endpoint-url http://localhost:4566

Scan table

aws dynamodb scan \ --table-name MyTable \ --endpoint-url http://localhost:4566

Query table

aws dynamodb query \ --table-name MyTable \ --key-condition-expression "Id = :id" \ --expression-attribute-values '{":id":{"S":"1"}}' \ --endpoint-url http://localhost:4566

Update item

aws dynamodb update-item \ --table-name MyTable \ --key '{"Id":{"S":"1"}}' \ --update-expression "SET #n = :val" \ --expression-attribute-names '{"#n":"Name"}' \ --expression-attribute-values '{":val":{"S":"Bob"}}' \ --endpoint-url http://localhost:4566

Delete item

aws dynamodb delete-item \ --table-name MyTable \ --key '{"Id":{"S":"1"}}' \ --endpoint-url http://localhost:4566
