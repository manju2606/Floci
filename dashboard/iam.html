Users

List users

aws iam list-users \ --endpoint-url http://localhost:4566

Create user

aws iam create-user \ --user-name alice \ --endpoint-url http://localhost:4566

Create access key

aws iam create-access-key \ --user-name alice \ --endpoint-url http://localhost:4566

Attach policy to user

aws iam attach-user-policy \ --user-name alice \ --policy-arn arn:aws:iam::aws:policy/AmazonS3FullAccess \ --endpoint-url http://localhost:4566

List user policies

aws iam list-attached-user-policies \ --user-name alice \ --endpoint-url http://localhost:4566

Delete user

aws iam delete-user \ --user-name alice \ --endpoint-url http://localhost:4566

Roles

List roles

aws iam list-roles \ --endpoint-url http://localhost:4566

Create role (EC2 trust)

aws iam create-role \ --role-name my-ec2-role \ --assume-role-policy-document '{ "Version":"2012-10-17", "Statement":[{"Effect":"Allow", "Principal":{"Service":"ec2.amazonaws.com"}, "Action":"sts:AssumeRole"}] }' \ --endpoint-url http://localhost:4566

Attach policy to role

aws iam attach-role-policy \ --role-name my-ec2-role \ --policy-arn arn:aws:iam::aws:policy/AmazonDynamoDBFullAccess \ --endpoint-url http://localhost:4566

List role policies

aws iam list-attached-role-policies \ --role-name my-ec2-role \ --endpoint-url http://localhost:4566

Delete role

aws iam delete-role \ --role-name my-ec2-role \ --endpoint-url http://localhost:4566

Policies

List customer policies

aws iam list-policies \ --scope Local \ --endpoint-url http://localhost:4566

Create policy

aws iam create-policy \ --policy-name my-s3-policy \ --policy-document '{ "Version":"2012-10-17", "Statement":[{"Effect":"Allow", "Action":["s3:GetObject","s3:PutObject"], "Resource":"arn:aws:s3:::my-bucket/*"}] }' \ --endpoint-url http://localhost:4566

Get policy

aws iam get-policy \ --policy-arn <policy-arn> \ --endpoint-url http://localhost:4566

Delete policy

aws iam delete-policy \ --policy-arn <policy-arn> \ --endpoint-url http://localhost:4566

Groups

List groups

aws iam list-groups \ --endpoint-url http://localhost:4566

Create group

aws iam create-group \ --group-name developers \ --endpoint-url http://localhost:4566

Add user to group

aws iam add-user-to-group \ --group-name developers \ --user-name alice \ --endpoint-url http://localhost:4566

Attach policy to group

aws iam attach-group-policy \ --group-name developers \ --policy-arn arn:aws:iam::aws:policy/ReadOnlyAccess \ --endpoint-url http://localhost:4566
